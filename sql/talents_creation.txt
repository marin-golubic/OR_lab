CREATE TABLE talents (
    talent_id SERIAL PRIMARY KEY,
    character_id INT NOT NULL,
    talent_name VARCHAR(100) NOT NULL,
    talent_type VARCHAR(30),

    FOREIGN KEY (character_id)
        REFERENCES characters(character_id)
);
